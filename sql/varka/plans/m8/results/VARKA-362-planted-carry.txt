# Replays the 128 by 128 to 256-bit column sum of VARKA-362 2.3 in Python with Java's 64-bit
# wrapping, on 32-bit limbs: sixteen limb products, each split into its low and high halves
# and added to its two columns, the carries propagated once at the end. Two sign forms: T, the
# operands' two's complement patterns multiplied unsigned with the sign correction folded into
# the high columns as limbs of ~y and ~x and a +1, and A, the magnitudes multiplied and the sign
# put back on the 256-bit result. The planted order sums two limb products of one column in one
# 64-bit word before splitting it: column 1's x0 * y1 + x1 * y0, QuestDB's a0 * m1 + a1 * m0,
# and column 3's x0 * y3 + x1 * y2, a middle column of four products.
import random

M = (1 << 32) - 1
W64 = 1 << 64
B = 10**38 - 1


def limbs(v, n):
    return [(v >> (32 * i)) & M for i in range(n)]


PLANTS = {None: (), 1: ((0, 1), (1, 0)), 3: ((0, 3), (1, 2))}


def column_sum(xl, yl, extra, planted):
    acc = [0] * 9
    pair = PLANTS[planted]
    for i in range(4):
        for j in range(4):
            if (i, j) in pair:
                continue
            p = xl[i] * yl[j]
            acc[i + j] += p & M
            acc[i + j + 1] += p >> 32
    if pair:
        (a, b), (c, d) = pair
        t = (xl[a] * yl[b] + xl[c] * yl[d]) % W64
        acc[a + b] += t & M
        acc[a + b + 1] += t >> 32
    for c, v in extra:
        acc[c] += v
    peak = max(acc)
    out = 0
    for c in range(8):
        acc[c + 1] += acc[c] >> 32
        out |= (acc[c] & M) << (32 * c)
    return out, peak


def signed256(v):
    return v - (1 << 256) if v >= 1 << 255 else v


def form_t(x, y, planted):
    xu, yu = x % (1 << 128), y % (1 << 128)
    extra = []
    if x < 0:
        extra += [(4 + i, l) for i, l in enumerate(limbs(~yu % (1 << 128), 4))] + [(4, 1)]
    if y < 0:
        extra += [(4 + i, l) for i, l in enumerate(limbs(~xu % (1 << 128), 4))] + [(4, 1)]
    v, peak = column_sum(limbs(xu, 4), limbs(yu, 4), extra, planted)
    return signed256(v), peak


def form_a(x, y, planted):
    v, peak = column_sum(limbs(abs(x), 4), limbs(abs(y), 4), [], planted)
    return (-v if (x < 0) != (y < 0) else v), peak


def cls(x, y):
    if x >= 0 and y >= 0:
        return "both >= 0"
    return "both negative" if x < 0 and y < 0 else "one negative"


draws = 500_000
