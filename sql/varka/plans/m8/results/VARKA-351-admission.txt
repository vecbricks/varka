    """RoundBase.dataType (mathExpressions.scala 1714-1730)."""
    ild = p - s + 1
    if k < 0:
        return min(max(ild, -k + 1), 38), 0
    ns = min(s, k)
    return min(ild + ns, 38), ns

def spark_round(u, s, k, mode, p2, s2):
    """RoundBase.nullSafeEval over a decimal; None where Decimal.set raises."""
    x = D(u).scaleb(-s)
    r = x.quantize(D(1).scaleb(-k), rounding=MODES[mode])
    v = int(r.scaleb(s2))
    return None if abs(v) >= 10 ** p2 else v
