# Python's int / int is correctly rounded to binary64, so u / 10**s is RN64(u * 10^-s), what
# BigDecimal.doubleValue returns (JDK 25, BigDecimal.java 3825-3862), for 128-bit u as well.
# Binary32 has no native rounding here, so rn(num, den, 24) rounds a rational by hand.
import math, random

def rn(num, den, prec, emin):
    """Round num/den (num != 0, den > 0) to nearest even with a prec-bit significand and
    minimum normal exponent emin; returns (sign, significand, exponent) with value
    sig * 2^exp, the significand reduced so equal values compare equal."""
    sign = -1 if num < 0 else 1
    num = abs(num)
    e = num.bit_length() - den.bit_length()
    if (num << max(0, -e)) < (den << max(0, e)):
        e -= 1
    # 2^e <= num/den < 2^(e+1)
    e = max(e, emin)
    sh = prec - 1 - e
    n, d = (num << sh, den) if sh >= 0 else (num, den << -sh)
    q, r = divmod(n, d)
    if 2 * r > d or (2 * r == d and q & 1):
        q += 1
    exp = -sh
    while q and q % 2 == 0:
        q //= 2
        exp += 1
    return (sign, q, exp)

def f64(num, den):
    return rn(num, den, 53, -1022)

def f32(num, den):
    return rn(num, den, 24, -126)

def bracket(a):
    """The two-pass inputs for a magnitude a < 2^127: (lo, hi, sh, exact) with
    lo = w' * 2^sh, hi = (w' + 1) * 2^sh and exact when no bit was shifted out."""
    if a < 2**64:
        return a, a, 0, True
    sh = a.bit_length() - 64
    w = a >> sh
    return w << sh, (w + 1) << sh, sh, (a & ((1 << sh) - 1)) == 0

def exact_rule(u, s, lo, hi):
    q = abs(u) // 10**s
    q = q if u >= 0 else -q
    return q if lo <= q <= hi else "overflow"

def double_rule(u, s, lo, hi):
    d = u / 10**s
    return math.trunc(d) if (math.floor(d) <= hi and math.ceil(d) >= lo) else "overflow"

def outcome_of_double(d, lo, hi):
    return math.trunc(d) if (math.floor(d) <= hi and math.ceil(d) >= lo) else "overflow"
