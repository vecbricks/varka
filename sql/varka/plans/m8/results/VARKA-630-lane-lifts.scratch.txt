package org.apache.spark.sql.catalyst.expressions.codegen;

import org.junit.jupiter.api.Test;
import org.apache.spark.sql.catalyst.expressions.codegen.varka.VarkaVectorIR.*;

class VarkaScratchLiftSuite {
  static Cond cond(int depth, LaneType lane) {
    Cond c = new Compare(CompareOp.LT, new ColumnRef(0, lane), new ColumnRef(1, lane));
    for (int i = 0; i < depth; i++) c = new And(c, new Not(c));
    return c;
  }

  @Test
  void time() {
    for (int d : new int[] {16, 24, 32, 48, 64}) {
      Cond c = cond(d, LaneType.SHORT);
      long t0 = System.nanoTime();
      Cond a = VarkaConditionCompiler.onIntLane(c);
      long t1 = System.nanoTime();
      var b = VarkaConditionCompiler.onLongLane(c).orElseThrow();
      long t2 = System.nanoTime();
      System.out.println("SCRATCH depth=" + d + " onIntLane_us=" + (t1 - t0) / 1000
          + " onLongLane_us=" + (t2 - t1) / 1000);
    }
  }
}
