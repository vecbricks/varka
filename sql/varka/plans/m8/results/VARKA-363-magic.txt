# The magic constants of the HALF_UP rescale down (VARKA-363 2.3): for a dividend magnitude
# n <= B, N = n >> (k - 1) and d = 5^k, the round-up constant c = ceil(2^L / d) with the least L
# meeting the sufficient condition e * Nmax < 2^L (e = c * d - 2^L), its width, the limb
# products of N * c on 32-bit limbs at three bounds, and a random and corner check of
# (floor(N * c / 2^L) + 1) >> 1 == HALF_UP(n / 10^k) for each k.
import random


def limbs(v):
    return max(1, -(-v.bit_length() // 32))


def magic(nmax, d):
    L = 0
    while True:
        c = -(-(1 << L) // d)
        e = c * d - (1 << L)
        if e * nmax < (1 << L):
            return L, c
        L += 1


def half_up(n, k):
    return (n + 10**k // 2) // 10**k


random.seed(363)
for name, B in (("128-bit dividend, |n| <= 10^38 - 1", 10**38 - 1),
                ("an equal-scale sum's magnitude, |n| <= 2 * 10^38 - 2", 2 * 10**38 - 2),
                ("256-bit dividend, |n| <= (10^38 - 1)^2", (10**38 - 1) ** 2)):
    print(f"== {name} ({B.bit_length()} bits)")
    print("   k   bits(Nmax)   L   bits(c)   limbs N x c = products   checked")
    for k in range(1, 40):
        nmax = B >> (k - 1)
        d = 5**k
        L, c = magic(nmax, d)
        ln, lc = limbs(nmax), limbs(c)
        draws = [random.randint(0, B) for _ in range(20000)]
        draws += [B, 0, 1, 10**k // 2 - 1, 10**k // 2, 10**k - 1, 10**k, 10**k + 10**k // 2]
        draws += [m * 10**k + 10**k // 2 + t for m in (1, B // 10**k - 1) for t in (-1, 0)]
        draws = [n for n in draws if 0 <= n <= B]
        ok = all((((n >> (k - 1)) * c >> L) + 1) >> 1 == half_up(n, k) for n in draws)
        print(f"  {k:2d}   {nmax.bit_length():3d}        {L:3d}   {c.bit_length():3d}       "
              f"{ln} x {lc} = {ln * lc:2d}               {len(draws)} {'ok' if ok else 'FAIL'}")
