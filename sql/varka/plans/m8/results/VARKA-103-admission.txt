// against Spark's semantics. Run with
//   java -Xmx8g -cp guava-33.6.0-jre.jar Varka103Admission.java
// (JDK 25, single-file source launch; Guava is the version Spark's pom.xml pins).
//
// Spark's answers are reproduced by the calls its code makes:
//   getDays(x) = (int) (x / D); getHours(x) = (byte) ((x % D) / H);
//   getMinutes(x) = (byte) ((x % H) / M); getSeconds(x) unscaled = x % M   (IntervalUtils.scala)
//   dt * k (integral) = Math.multiplyExact(x, k)                   (intervalExpressions.scala)
//   dt / k (integral) = LongMath.divide(x, k, HALF_UP)
//   dt * f, dt / f (floating) = DoubleMath.roundToLong((double) x * f or / f, HALF_UP)
//   dt * c, dt / c (decimal) = BigDecimal multiply under MathContext(39, DOWN), or divide to
//     scale 39 DOWN, then setScale(0, HALF_UP).longValueExact()      (Decimal.scala)
//   make_dt_interval = addExact(secs, multiplyExact(days, D)), + hours * H, + mins * M
// The constructions use only + - * on longs (wrapping), shifts, ands, compares and Java's /
// where the plan lowers it to a bounded ConstDivide. These are sampled checks, not proofs.

  /** 338's floorDivLong: an arithmetic shift by 13, the down step, a ConstDivide by ODD. */
  static long floorDivLong(long x, long[] maxShifted) {
    long y = x >> 13;
    long down = (y >> 63) & (1 - ODD);
    long z = y + down;
    long a = Math.abs(z);
    if (a > maxShifted[0]) maxShifted[0] = a;
    return z / ODD; // ConstDivide truncates; exact while |z| <= 2^52
  }

  /** 332's 2.4 truncation: the floor of a biased dividend. */
  static long truncDay(long x, long[] maxShifted) {
    long bias = (x >> 63) & (D - 1);
    return floorDivLong(x + bias, maxShifted);
  }

  /** The row text's form: the magnitude, its floor, the sign restored; shift as given. */
  static long truncDayMagnitude(long x, boolean logicalShift) {
    long s = x >> 63;
    long mag = (x ^ s) - s; // wraps at MIN
    long y = logicalShift ? mag >>> 13 : mag >> 13;
    long q = y / ODD;
    return (q ^ s) - s;
  }

  static long halfUpConst(long x, long k) {
    long q = x / k;
    long r = x - q * k;
    // ceil(|k| / 2) without overflow: |k| >> 1 plus its low bit; 2^62 for MIN.
    long ak = Math.abs(k);
    long half = k == Long.MIN_VALUE ? 1L << 62 : (ak >> 1) + (ak & 1);
    long ar = Math.abs(r);
    int away = (k > 0) == (x >= 0) ? 1 : -1;
    return ar >= half && r != 0 ? q + away : q;
  }

  /** dt / k for k dividing D, over every long: the day split, then the bounded division. */
  static long halfUpDaySplit(long x, long k, long[] maxShifted) {
    long qd = truncDay(x, maxShifted);
    long rd = x - qd * D;
    long qk = qd * (D / k) + rd / k; // rd / k: ConstDivide bounded by D
    long rk = rd % k;
    long half = (Math.abs(k) + 1) / 2;
    int away = (k > 0) == (x >= 0) ? 1 : -1;
    return Math.abs(rk) >= half && rk != 0 ? qk + away : qk;
  }

  /** Row 342's 2.4 form for the floating factor, the guard folded in as a null result. */
  static Long halfUpDouble(double p) {
    if (!(p >= -0x1p63 && p < 0x1p63)) return null;
    long t = (long) p;
    double f = p - (double) t;
    return t + (f >= 0.5 ? 1 : f <= -0.5 ? -1 : 0);
  }

  static Long sparkRound(double p) {

    catch (ArithmeticException e) { return null; }
  }

  static Long sparkMakeDt(int days, int hours, int mins, long secs) {
    long micros = secs;
    try {

      micros = Math.addExact(micros, Math.multiplyExact(hours, H));
      micros = Math.addExact(micros, Math.multiplyExact(mins, M));
      return micros;
    } catch (ArithmeticException e) { return null; }
  }

  /** The planned make_dt_interval: days guarded, three checked adds in Spark's order. */
  static Long planMakeDt(int days, int hours, int mins, long secs) {
    if (days < -106_751_991 || days > 106_751_991) return null;
    long s1 = secs + (long) days * D, s2, s3;
    if (((secs ^ s1) & (((long) days * D) ^ s1)) < 0) return null;
    long p2 = (long) hours * H;
    s2 = s1 + p2;
    if (((s1 ^ s2) & (p2 ^ s2)) < 0) return null;
    long p3 = (long) mins * M;
    s3 = s2 + p3;
    if (((s2 ^ s3) & (p3 ^ s3)) < 0) return null;
    return s3;
  }

  static long[] guard(long k) {
    BigInteger min = BigInteger.valueOf(Long.MIN_VALUE), max = BigInteger.valueOf(Long.MAX_VALUE);
    BigInteger kb = BigInteger.valueOf(k);
    BigInteger a, b;
    if (k > 0) { a = ceilDiv(min, kb); b = floorDiv(max, kb); }
    else { a = ceilDiv(max, kb); b = floorDiv(min, kb); }
    BigInteger lo = a.max(min), hi = b.min(max);
    return new long[] {lo.longValueExact(), hi.longValueExact()};
  }

  static BigInteger floorDiv(BigInteger a, BigInteger b) {
    BigInteger[] qr = a.divideAndRemainder(b);
    return qr[1].signum() != 0 && (qr[1].signum() != b.signum()) ? qr[0].subtract(BigInteger.ONE)
        : qr[0];
  }

  static BigInteger ceilDiv(BigInteger a, BigInteger b) {
    return floorDiv(a.negate(), b).negate();
  }

  static boolean multiplyRaises(long x, long k) {
    try { Math.multiplyExact(x, k); return false; } catch (ArithmeticException e) { return true; }
  }

  static Long sparkDecimalTimes(long x, BigDecimal c) {
    try {
      return new BigDecimal(x).multiply(c, new MathContext(39, RoundingMode.DOWN))
          .setScale(0, RoundingMode.HALF_UP).longValueExact();
    } catch (ArithmeticException e) { return null; }
  }

  static Long sparkDecimalDiv(long x, BigDecimal c) {
    try {
      return new BigDecimal(x).divide(c, 39, RoundingMode.DOWN)
          .setScale(0, RoundingMode.HALF_UP).longValueExact();
    } catch (ArithmeticException e) { return null; }
  }

    AtomicLong n = new AtomicLong();
    s.forEach(x -> { f.accept(x); n.incrementAndGet(); });
