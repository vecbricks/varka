// VARKA-345 admission checks: what the Vector API's math operators bind to on this host, what
// they compute outside C2, what a step count admits against a multiple of Math.ulp, and what a
// lane the kernel does not use costs a library call. Run (JDK 25, single-file source launch) as
//   java --add-modules jdk.incubator.vector -Djdk.incubator.vector.DEBUG=true \
//     -XX:CompileCommand=quiet -XX:CompileCommand=exclude,Varka345Admission::cold* \
//     Varka345Admission.java
// The DEBUG lines the JDK prints while part 1 runs are its own (VectorMathLibrary.constructEntry).
import java.util.*;
import java.util.function.*;
import jdk.incubator.vector.*;

public class Varka345Admission {
  static final VectorOperators.Unary[] UNARY = {
    VectorOperators.SIN, VectorOperators.COS, VectorOperators.TAN, VectorOperators.ASIN,
    VectorOperators.ACOS, VectorOperators.ATAN, VectorOperators.EXP, VectorOperators.LOG,
    VectorOperators.LOG10, VectorOperators.CBRT, VectorOperators.SINH, VectorOperators.COSH,
    VectorOperators.TANH, VectorOperators.EXPM1, VectorOperators.LOG1P};
  static final VectorOperators.Binary[] BINARY = {
    VectorOperators.ATAN2, VectorOperators.POW, VectorOperators.HYPOT};
  static final VectorSpecies<Double> S = DoubleVector.SPECIES_PREFERRED;
  static final int N = 1 << 16;

  // ---- part 2: one operator, a compile-time constant, in a hot loop and in an excluded one ----
  static void hotSin(double[] in, double[] out) {
    for (int i = 0; i < in.length; i += S.length()) {
      DoubleVector.fromArray(S, in, i).lanewise(VectorOperators.SIN).intoArray(out, i);
    }
  }
  static void coldSin(double[] in, double[] out) {
    for (int i = 0; i < in.length; i += S.length()) {
      DoubleVector.fromArray(S, in, i).lanewise(VectorOperators.SIN).intoArray(out, i);
    }
  }
  static void hotExp(double[] in, double[] out) {
    for (int i = 0; i < in.length; i += S.length()) {
      DoubleVector.fromArray(S, in, i).lanewise(VectorOperators.EXP).intoArray(out, i);
    }
  }
  static void coldExp(double[] in, double[] out) {
    for (int i = 0; i < in.length; i += S.length()) {
      DoubleVector.fromArray(S, in, i).lanewise(VectorOperators.EXP).intoArray(out, i);
    }
  }

  // ---- part 4: the cost of an argument in a lane the kernel does not use ----
  static void sinLoop(double[] in, double[] out) {
    for (int i = 0; i < in.length; i += S.length()) {
      DoubleVector.fromArray(S, in, i).lanewise(VectorOperators.SIN).intoArray(out, i);
    }
  }
  static void sinMasked(double[] in, double[] out, VectorMask<Double> m) {
    for (int i = 0; i < in.length; i += S.length()) {
      DoubleVector.fromArray(S, in, i).lanewise(VectorOperators.SIN, m).intoArray(out, i);
    }
  }
  static void sinBlended(double[] in, double[] out, boolean[] unused) {
    for (int i = 0; i < in.length; i += S.length()) {
      VectorMask<Double> off = VectorMask.fromArray(S, unused, i);
      DoubleVector.fromArray(S, in, i).blend(0.5, off).lanewise(VectorOperators.SIN)
          .intoArray(out, i);
    }
  }
  static void tanLoop(double[] in, double[] out) {
    for (int i = 0; i < in.length; i += S.length()) {
      DoubleVector.fromArray(S, in, i).lanewise(VectorOperators.TAN).intoArray(out, i);
    }
  }
  static void expLoop(double[] in, double[] out) {
    for (int i = 0; i < in.length; i += S.length()) {
      DoubleVector.fromArray(S, in, i).lanewise(VectorOperators.EXP).intoArray(out, i);
    }
  }
  static void logLoop(double[] in, double[] out) {
    for (int i = 0; i < in.length; i += S.length()) {
      DoubleVector.fromArray(S, in, i).lanewise(VectorOperators.LOG).intoArray(out, i);
    }
  }

  static double nsPerElement(Runnable r) {
    for (int w = 0; w < 400; w++) r.run();
    double best = Double.MAX_VALUE;
    for (int t = 0; t < 9; t++) {
      long t0 = System.nanoTime();
      for (int k = 0; k < 40; k++) r.run();
      best = Math.min(best, (System.nanoTime() - t0) / (40.0 * N));
    }
    return best;
  }

  static long steps(double a, double b) {
    return Math.abs(Double.doubleToRawLongBits(a) - Double.doubleToRawLongBits(b));
  }

  public static void main(String[] args) {
    System.out.println("host: java.runtime.version=" + System.getProperty("java.runtime.version")
        + " os.arch=" + System.getProperty("os.arch") + " preferred species=" + S
        + " library=" + System.getProperty("jdk.incubator.vector.VectorMathLibrary", "default"));

    System.out.println("\n1. The binding: each operator once at 512, 256 and 128 bits (the DEBUG"
        + " lines are the JDK's; an operator with no line has no symbol at that species)");
    for (VectorSpecies<Double> sp : List.of(DoubleVector.SPECIES_512, DoubleVector.SPECIES_256,
        DoubleVector.SPECIES_128)) {
      if (sp.vectorBitSize() > S.vectorBitSize()) continue;
      System.out.println("-- " + sp);
      DoubleVector v = DoubleVector.broadcast(sp, 0.5);
      for (VectorOperators.Unary op : UNARY) v.lanewise(op);
      for (VectorOperators.Binary op : BINARY) v.lanewise(op, v);
    }

    System.out.println("\n2. Outside C2: the same loop hot and excluded from compilation, "
        + N + " inputs");
    Random r = new Random(345);
    double[] in = new double[N], hot = new double[N], cold = new double[N];
    for (int i = 0; i < N; i++) in[i] = (r.nextDouble() - 0.5) * 20;
    for (int k = 0; k < 3000; k++) hotSin(in, hot);
    coldSin(in, cold);
    int hotVsMath = 0, coldVsMath = 0;
    for (int i = 0; i < N; i++) {
      if (hot[i] != Math.sin(in[i])) hotVsMath++;
      if (Double.doubleToRawLongBits(cold[i]) != Double.doubleToRawLongBits(Math.sin(in[i])))
        coldVsMath++;
    }
    System.out.printf("SIN hot (C2):      %6d lanes differ from Math.sin%n", hotVsMath);
    System.out.printf("SIN excluded:      %6d lanes differ from Math.sin%n", coldVsMath);
    for (int i = 0; i < N; i++) in[i] = (r.nextDouble() - 0.5) * 40;
    for (int k = 0; k < 3000; k++) hotExp(in, hot);
    coldExp(in, cold);
    int hotVsStrict = 0, coldVsStrict = 0, coldVsMathExp = 0;
    for (int i = 0; i < N; i++) {
      if (hot[i] != StrictMath.exp(in[i])) hotVsStrict++;
      if (cold[i] != StrictMath.exp(in[i])) coldVsStrict++;
      if (Double.doubleToRawLongBits(cold[i]) != Double.doubleToRawLongBits(Math.exp(in[i])))
        coldVsMathExp++;
    }
    System.out.printf("EXP hot (C2):      %6d lanes differ from StrictMath.exp%n", hotVsStrict);
    System.out.printf("EXP excluded:      %6d lanes differ from StrictMath.exp,"
        + " %d from Math.exp%n", coldVsStrict, coldVsMathExp);

    System.out.println("\n3. What a bound of one admits: steps against |a - r| <= Math.ulp(r)");
    double one = 1.0;
    double below2 = Math.nextDown(Math.nextDown(one));
    System.out.printf("r = 1.0, a = two doubles below: steps %d, |a - r| <= ulp(r): %b%n",
        steps(below2, one), Math.abs(below2 - one) <= Math.ulp(one));
    double inf = Double.POSITIVE_INFINITY;
    System.out.printf("r = +inf, a = 1.0: |a - r| <= ulp(r): %b%n",
        Math.abs(1.0 - inf) <= Math.ulp(inf));
    System.out.printf("r = +0.0, a = -0.0: |a - r| <= ulp(r): %b, raw-bit steps %d%n",
        Math.abs(-0.0 - 0.0) <= Math.ulp(0.0), steps(-0.0, 0.0));
    System.out.printf("r = Double.MAX_VALUE, a = +inf: raw-bit steps %d%n",
        steps(inf, Double.MAX_VALUE));
    System.out.printf("r = Double.MIN_VALUE, a = -Double.MIN_VALUE: raw-bit steps %d%n",
        steps(-Double.MIN_VALUE, Double.MIN_VALUE));

    System.out.println("\n4. A lane the kernel does not use, C2 hot loops, laptop, ns per element"
        + " (minimum of 9 timings of 40 passes over " + N + ")");
    double[] ordinary = new double[N], out = new double[N];
    for (int i = 0; i < N; i++) ordinary[i] = 0.25 + (i % 1000) * 1e-3;
    BiFunction<double[], Double, double[]> oneInEight = (base, special) -> {
      double[] a = base.clone();
      for (int i = 0; i < N; i += 8) a[i] = special;
      return a;
    };
    double[] huge = oneInEight.apply(ordinary, 1e300);
    double[] nan = oneInEight.apply(ordinary, Double.NaN);
    double[] overflow = oneInEight.apply(ordinary, 800.0);
    double[] subnormal = oneInEight.apply(ordinary, Double.MIN_VALUE);
    double[] negative = oneInEight.apply(ordinary, -1.0);
    System.out.printf("SIN ordinary               %6.3f%n", nsPerElement(() -> sinLoop(ordinary, out)));
    System.out.printf("SIN 1 lane in 8 at 1e300   %6.3f%n", nsPerElement(() -> sinLoop(huge, out)));
    System.out.printf("SIN 1 lane in 8 NaN        %6.3f%n", nsPerElement(() -> sinLoop(nan, out)));
    System.out.printf("TAN ordinary               %6.3f%n", nsPerElement(() -> tanLoop(ordinary, out)));
    System.out.printf("TAN 1 lane in 8 at 1e300   %6.3f%n", nsPerElement(() -> tanLoop(huge, out)));
    System.out.printf("EXP ordinary               %6.3f%n", nsPerElement(() -> expLoop(ordinary, out)));
    System.out.printf("EXP 1 lane in 8 at 800     %6.3f%n", nsPerElement(() -> expLoop(overflow, out)));
    System.out.printf("LOG ordinary               %6.3f%n", nsPerElement(() -> logLoop(ordinary, out)));
    System.out.printf("LOG 1 lane in 8 subnormal  %6.3f%n", nsPerElement(() -> logLoop(subnormal, out)));
    System.out.printf("LOG 1 lane in 8 at -1      %6.3f%n", nsPerElement(() -> logLoop(negative, out)));
    boolean[] unused = new boolean[N];
    for (int i = 0; i < N; i += 8) unused[i] = true;
    boolean[] noneUnused = new boolean[N];
    System.out.printf("SIN ordinary, the blend to 0.5 by a loaded mask, no lane set  %6.3f%n",
        nsPerElement(() -> sinBlended(ordinary, out, noneUnused)));
    System.out.printf("SIN 1 lane in 8 at 1e300, that lane blended to 0.5 first     %6.3f%n",
        nsPerElement(() -> sinBlended(huge, out, unused)));
    System.out.printf("SIN 1 lane in 8 NaN, that lane blended to 0.5 first          %6.3f%n",
        nsPerElement(() -> sinBlended(nan, out, unused)));
    VectorMask<Double> none = S.maskAll(false);
    System.out.printf("SIN masked, no lane set, 1 lane in 8 at 1e300 %6.3f%n",
        nsPerElement(() -> sinMasked(huge, out, none)));
    sinMasked(huge, out, none);
    int kept = 0;
    for (int i = 0; i < N; i++) if (Double.doubleToRawLongBits(out[i]) == Double.doubleToRawLongBits(huge[i])) kept++;
    System.out.printf("SIN masked, no lane set: %d of %d lanes are the input unchanged%n", kept, N);
  }
}
