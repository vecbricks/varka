// semantics. Run with `java Varka346Admission.java` (JDK 25, single-file source launch).

// Cast.scala castToDecimal, arithmetic.scala Remainder, MathUtils.pmod):
//   round/bround/truncate over DOUBLE: BigDecimal.valueOf(d).setScale(s, mode).doubleValue()
//   over FLOAT: BigDecimal.valueOf((double) f).setScale(s, mode).floatValue()
//   CAST(d AS DECIMAL(p, s)): the unscaled value of BigDecimal.valueOf(d).setScale(s, HALF_UP)
//   x % y: Java's %; pmod: r = x % n; r < 0 ? (r + n) % n : r

      while (!(new BigDecimal(Math.scalb(1.0, e - 53)).compareTo(BigDecimal.ONE.movePointLeft(s + 1)) < 0)) e--;
      LE[s] = e;
      L[s] = Math.scalb(1.0, e);
      BigDecimal k = BigDecimal.ONE.movePointRight(16 - s);
      double kd = k.doubleValue();

      K[s] = Math.min(Math.scalb(1.0, 52), kd);
    }
  }

  static final String[] MN = {"round (HALF_UP)", "bround (HALF_EVEN)", "truncate (DOWN)"};
  static final double BAND = Double.longBitsToDouble(0x7ff8dead0000beefL); // marker only

  // The magnitude n = mode-rounding of the shortest decimal of a = |d| at scale s, for a < L[s].
  static double nOf(double a, int s, int mode) {
    double p = P[s];
    double y = a * p;
    if (mode == DOWN) {
      double m = Math.rint(y);
      double t = m / p;                       // Clinger: m < 2^53, p exact
      return a < t ? m - 1 : m;
    }
    double k = Math.floor(y);
    double t = (2 * k + 1) / P2[s];           // Clinger: 2k + 1 < 2^53, 2 * 10^s exact
    boolean up = mode == HALF_UP ? a >= t : (a > t || (a == t && (((long) k) & 1) == 1));
    return up ? k + 1 : k;
  }

  static double roundD(double d, int s, int mode) {
    if (Double.isNaN(d) || Double.isInfinite(d)) return d;
    double a = Math.abs(d);
    if (a >= K[s]) return d;
    if (a >= L[s]) return BAND;
    double q = nOf(a, s, mode) / P[s];
    long sign = Double.doubleToRawLongBits(d) & Long.MIN_VALUE;
    return Double.longBitsToDouble(Double.doubleToRawLongBits(q) | sign) + 0.0;
  }

  static double sparkD(double d, int s, int mode) {
    if (Double.isNaN(d) || Double.isInfinite(d)) return d;
    return BigDecimal.valueOf(d).setScale(s, RM[mode]).doubleValue();
  }

  // RN32 of the exact n / p by round-to-odd at 53 bits, then one narrowing.
  static float toFloatOdd(double n, double p) {
    double q = n / p;
    double r = Math.fma(-q, p, n);
    long b = Double.doubleToRawLongBits(q);
    if (r != 0 && (b & 1) == 0) b += (r > 0) == (q > 0) ? 1 : -1;
    return (float) Double.longBitsToDouble(b);
  }

  static float roundF(float f, int s, int mode) {
    if (Float.isNaN(f) || Float.isInfinite(f)) return f;
    double d = f;
    double a = Math.abs(d);
    if (a >= K[s]) return f;
    if (a >= L[s]) return Float.NaN; // band marker for floats
    float q = toFloatOdd(nOf(a, s, mode), P[s]);
    int sign = Float.floatToRawIntBits(f) & Integer.MIN_VALUE;
    return Float.intBitsToFloat(Float.floatToRawIntBits(q) | sign) + 0.0f;
  }

  static float sparkF(float f, int s, int mode) {
    if (Float.isNaN(f) || Float.isInfinite(f)) return f;
    return BigDecimal.valueOf((double) f).setScale(s, RM[mode]).floatValue();
  }

  // The one-FMA remainder on magnitudes; DECLINE where |x| >= |y| and |x / y| >= 2^53.
  static final double DECLINE = Double.longBitsToDouble(0x7ff8dead0000f00dL);
  static double rem(double x, double y) {
    double ax = Math.abs(x), ay = Math.abs(y);
    long sx = Double.doubleToRawLongBits(x) & Long.MIN_VALUE;
    if (ax < ay) return x;                    // also every finite x over an infinite y
    double qd = ax / ay;
    if (ax < Double.POSITIVE_INFINITY && qd >= 0x1p53) return DECLINE;
    double q = Math.floor(qd);
    double r = Math.fma(-q, ay, ax);
    if (r < 0) r = r + ay;
    return Double.longBitsToDouble(Double.doubleToRawLongBits(r) | sx);
  }

  static double pmodV(double x, double n) {
    double r = rem(x, n);
    if (r == DECLINE || Double.doubleToRawLongBits(r) == Double.doubleToRawLongBits(DECLINE)) return r;
    return r < 0 ? rem(r + n, n) : r;
  }

  static double pmodSpark(double a, double n) {
    double r = a % n;
    return r < 0 ? (r + n) % n : r;
  }

  static boolean same(double x, double y) {
    if (Double.isNaN(x) && Double.isNaN(y)) return true;
    return Double.doubleToRawLongBits(x) == Double.doubleToRawLongBits(y);
  }

  static boolean isMarker(double v, double m) {
    return Double.doubleToRawLongBits(v) == Double.doubleToRawLongBits(m);
  }

  static double randIn(SplittableRandom r, int e) { // a random double with exponent e (normal or subnormal)
    if (e < -1022) return Double.longBitsToDouble(r.nextLong(1L, 1L << 52));
    long mant = r.nextLong(1L << 52);
    return Double.longBitsToDouble(((long) (e + 1023) << 52) | mant);
  }
