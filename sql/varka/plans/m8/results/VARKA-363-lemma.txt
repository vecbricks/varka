VARKA-363: Spark's MathContext(39, DOWN) product then HALF_UP, against one HALF_UP of the exact product (30 October 2026, laptop).
Command: python3 -I lemma.py (the script below). Output:

400000 draws over cut shapes, seed 45786: 130510 values and 62548 overflows agree; the 39-digit truncation dropped digits in 105031; no disagreement

Script:

# Spark's Decimal.* multiplies under MathContext(39, DOWN) (Decimal.scala 524, 597) and
# changePrecision then rounds HALF_UP to the result scale and fails past the result precision
# (Decimal.scala 387-470). Replays both orders with Python's decimal module over random operand
# pairs of every cut shape (both settings of allowPrecisionLoss) and checks that the two-step
# answer, value or overflow, equals one HALF_UP of the exact product.
import random
from decimal import Decimal, Context, ROUND_DOWN, ROUND_HALF_UP

MAXP = 38
DOWN39 = Context(prec=39, rounding=ROUND_DOWN, Emax=999999, Emin=-999999)
EXACT = Context(prec=200, rounding=ROUND_HALF_UP, Emax=999999, Emin=-999999)


def adjust(p, s):
    if p <= MAXP:
        return p, s
    return MAXP, max(MAXP - (p - s), min(s, 6))


def bounded(p, s):
    return min(p, MAXP), min(s, MAXP)


def to_precision(v, p, s):
    q = v.quantize(Decimal(1).scaleb(-s), rounding=ROUND_HALF_UP, context=EXACT)
    return None if len(q.as_tuple().digits) > p and q != 0 else q


def draw(p, s):
    kind = random.random()
    if kind < 0.2:
        u = 10**p - 1 - random.randint(0, 10)
    elif kind < 0.4:
        u = random.randint(0, 10 ** random.randint(0, p) - 1)
    else:
        u = random.randint(0, 10**p - 1)
    u = -u if random.random() < 0.5 else u
    return Decimal(u).scaleb(-s)


random.seed(45786)
shapes = [(p1, s1, p2, s2) for p1 in range(1, 39) for s1 in range(0, p1 + 1)
          for p2 in range(1, 39) for s2 in range(0, p2 + 1)]
draws = 400000
checked = {"value": 0, "overflow": 0}
truncated = 0
for _ in range(draws):
    p1, s1, p2, s2 = random.choice(shapes)
    rule = adjust if random.random() < 0.5 else bounded
    pr, sr = rule(p1 + p2 + 1, s1 + s2)
    if sr == s1 + s2:
        continue
    x, y = draw(p1, s1), draw(p2, s2)
    exact = x.fma(y, Decimal(0), context=EXACT)
    two_step = to_precision(DOWN39.multiply(x, y), pr, sr)
    one = to_precision(exact, pr, sr)
    assert two_step == one, (p1, s1, p2, s2, pr, sr, x, y, two_step, one)
    if DOWN39.multiply(x, y) != exact:
        truncated += 1
    checked["overflow" if one is None else "value"] += 1
print(f"{draws} draws over cut shapes, seed 45786: {checked['value']} values and "
      f"{checked['overflow']} overflows agree; the 39-digit truncation dropped digits in "
      f"{truncated}; no disagreement")
