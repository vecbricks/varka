public class RintNaN {
  static final long[] IN = {0x7FF8000000000000L, 0xFFF8000000000000L, 0x7FF0000000000001L,
      0xFFF0000000000001L, 0x7FF4000000000000L, 0xFFF4000000000000L, 0x7FF8000000000001L};
  static double rint(double x) { return Math.rint(x); }
  public static void main(String[] a) {
    double[] xs = new double[IN.length];
    for (int i = 0; i < IN.length; i++) xs[i] = Double.longBitsToDouble(IN[i]);
    long[] out = new long[IN.length];
    for (int rep = 0; rep < 200_000; rep++)
      for (int i = 0; i < xs.length; i++) out[i] = Double.doubleToRawLongBits(rint(xs[i]));
    for (int i = 0; i < IN.length; i++)
      System.out.printf("%016x -> %016x%n", IN[i], out[i]);
  }
}
