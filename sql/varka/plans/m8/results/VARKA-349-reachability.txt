package org.apache.spark.sql.execution.benchmark

import org.apache.spark.SparkFunSuite
import org.apache.spark.sql.{Dataset, Encoders, SparkSession}
import org.apache.spark.sql.catalyst.expressions._
import org.apache.spark.sql.catalyst.expressions.aggregate.DeclarativeAggregate
import org.apache.spark.sql.execution.{FilterExec, ProjectExec, SerializeFromObjectExec,
  SparkPlan, VarkaColumnarToRowExec, VarkaProjectExec}
import org.apache.spark.sql.execution.aggregate.BaseAggregateExec
import org.apache.spark.sql.execution.window.WindowExec
import org.apache.spark.sql.internal.SQLConf
import org.apache.spark.sql.types.DecimalType

/** Throwaway: VARKA-349 step 1, where Spark evaluates the decimal internals. */
class Varka349ReachabilitySuite extends SparkFunSuite {

  private def interesting(e: Expression): Boolean = e match {
    case _: Add | _: Subtract | _: Multiply | _: Divide => true
    case _: UnscaledValue | _: MakeDecimal | _: CheckOverflow => true
    case c: Cast => c.dataType.isInstanceOf[DecimalType] ||
      c.child.dataType.isInstanceOf[DecimalType]
    case _ => false
  }

  private def found(where: String, es: Seq[Expression]): Seq[String] =
    es.flatMap(_.collect { case e if interesting(e) => e }).map { e =>
      s"    [$where] ${e.prettyName}: ${e.dataType.simpleString} <- " +
        s"${e.children.map(_.dataType.simpleString).mkString(", ")}   ${e.sql.take(160)}"
    }

  private def nodeReport(p: SparkPlan): Seq[String] = p match {
    case a: BaseAggregateExec =>
      val name = a.getClass.getSimpleName
      val aggs = a.aggregateExpressions.flatMap { ae =>
        val head = s"  $name aggregate ${ae.mode} ${ae.aggregateFunction}: " +
          s"${ae.aggregateFunction.dataType.simpleString}"
        val body = ae.aggregateFunction match {
          case d: DeclarativeAggregate =>
            Seq(s"    buffer: ${d.aggBufferAttributes.map(x =>
              s"${x.name}:${x.dataType.simpleString}").mkString(", ")}") ++
              d.updateExpressions.map(e => s"    update: ${e.sql} : ${e.dataType.simpleString}") ++
              d.mergeExpressions.map(e => s"    merge: ${e.sql} : ${e.dataType.simpleString}") ++
              Seq(s"    evaluate: ${d.evaluateExpression.sql} : " +
                s"${d.evaluateExpression.dataType.simpleString}") ++
              found(s"$name.update", d.updateExpressions) ++
              found(s"$name.merge", d.mergeExpressions) ++
              found(s"$name.evaluate", Seq(d.evaluateExpression))
          case other => Seq(s"    (imperative ${other.getClass.getSimpleName})")
        }
        head +: body
      }
      Seq(s"  $name groupingExpressions: ${a.groupingExpressions.map(_.sql).mkString(", ")}") ++
        aggs ++
        a.resultExpressions.map(e => s"  $name result: ${e.sql} : ${e.dataType.simpleString}") ++
        found(s"$name.input", a.aggregateExpressions.flatMap(_.aggregateFunction.children)) ++
        found(s"$name.result", a.resultExpressions)
    case w: WindowExec =>
      w.windowExpression.map(e => s"  WindowExec windowExpression: ${e.sql} : " +
        s"${e.dataType.simpleString}") ++ found("WindowExec", w.windowExpression)
    case s: SerializeFromObjectExec =>
      s.serializer.map(e => s"  SerializeFromObjectExec serializer: ${e.sql} : " +
        s"${e.dataType.simpleString}") ++ found("SerializeFromObjectExec", s.serializer)
    case pr: ProjectExec =>
      Seq(s"  ProjectExec: ${pr.projectList.map(_.sql).mkString(", ")}") ++
        found("ProjectExec", pr.projectList)
    case pr: VarkaProjectExec =>
      Seq(s"  VarkaProjectExec: ${pr.projectList.map(_.sql).mkString(", ")}") ++
        found("VarkaProjectExec", pr.projectList)
    case pr: VarkaColumnarToRowExec =>
      Seq(s"  VarkaColumnarToRowExec: ${pr.projectList.map(_.sql).mkString(", ")}") ++
        found("VarkaColumnarToRowExec", pr.projectList)
    case f: FilterExec =>
      Seq(s"  FilterExec: ${f.condition.sql}") ++ found("FilterExec", Seq(f.condition))
    case other =>
      found(other.getClass.getSimpleName, other.expressions)
  }

  private def allNodes(p: SparkPlan): Seq[SparkPlan] =
    p +: (p.children ++ p.subqueries).flatMap(allNodes)

  private def report(title: String, df: Dataset[_], plain: SparkSession => Dataset[_]): Unit = {
    df.collect()
    val qe = df.queryExecution
    val out = new StringBuilder
    out ++= s"==== $title\n"
    out ++= s"-- output schema: ${df.schema.simpleString}\n"
    out ++= "-- optimized logical plan\n" + qe.optimizedPlan.treeString(verbose = true)
    out ++= "-- interesting expressions per logical node\n"
    qe.optimizedPlan.foreach { n =>
      found(n.nodeName, n.expressions).foreach(l => out ++= l + "\n")
    }
    out ++= "-- physical plan (Varka session)\n" + qe.executedPlan.treeString
    out ++= "-- per physical node\n"
    allNodes(qe.executedPlan).foreach { n =>
      nodeReport(n).foreach(l => out ++= l + "\n")
    }
    val varkaNodes = allNodes(qe.executedPlan).map(_.nodeName).filter(_.startsWith("Varka"))
    out ++= s"-- Varka nodes in the plan: ${if (varkaNodes.isEmpty) "none"
      else varkaNodes.mkString(", ")}\n"
    val a = qe.executedPlan.treeString.replaceAll("#\\d+", "#").replaceAll("plan_id=\\d+", "plan_id=")
      .replaceAll("\\$\\$Lambda[^,\\s]*", "\\$\\$Lambda")
    spark.conf.set(SQLConf.VARKA_ENABLED.key, "false")
    val b = try {
      val plainDf = plain(spark)
      plainDf.collect()
      plainDf.queryExecution.executedPlan.treeString.replaceAll("#\\d+", "#").replaceAll("plan_id=\\d+", "plan_id=")
      .replaceAll("\\$\\$Lambda[^,\\s]*", "\\$\\$Lambda")
    } finally spark.conf.set(SQLConf.VARKA_ENABLED.key, "true")
    out ++= s"-- plain session (spark.sql.codegen.varka.enabled=false): " +
      s"${if (a == b) "same physical tree up to expression ids" else "differs:\n" + b}\n"
    // scalastyle:off println
    println(out.toString)
    // scalastyle:on println
  }

  private var spark: SparkSession = _

  test("VARKA-349 reachability") {
    spark = VarkaArrowSessions.createSession("varka-349-reachability", true, false, "local[1]")
    try {
      def decimal(long: String, precision: Int): String =
        s"CAST(CAST($long % ${BigInt(10).pow(precision)} AS DECIMAL(${precision + 1}, 0)) * " +
          s"0.01 AS DECIMAL($precision, 2))"
      spark.sql(
        s"""SELECT id AS rid, id % 4 AS k, l,
           |       ${decimal("l", 8)} AS d8, ${decimal("l", 9)} AS d9,
           |       ${decimal("l", 18)} AS d18
           |FROM (SELECT id, CASE WHEN id % 7 = 3 THEN NULL
           |                      ELSE (id * 2654435761) - 1000000000000 END AS l
           |      FROM range(0, 4096))""".stripMargin).createOrReplaceTempView("t")
      VarkaArrowSessions.cache(spark, "t")
      // scalastyle:off println
      println(s"==== fixture schema: ${spark.table("t").schema.simpleString}")
      println(s"==== Varka session conf ${SQLConf.VARKA_ENABLED.key}=" +
        spark.conf.get(SQLConf.VARKA_ENABLED.key))
      // scalastyle:on println

      val window = "OVER (PARTITION BY k ORDER BY rid " +
        "ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW)"
      val queries = Seq(
        "Q1 sum(d8)" -> "SELECT sum(d8) FROM t",
        "Q1g sum(d8) GROUP BY k" -> "SELECT k, sum(d8) FROM t GROUP BY k",
        "Q1p sum(d8) GROUP BY k over three partitions (partial and final)" ->
          "SELECT k, sum(d8) FROM (SELECT /*+ REPARTITION(3) */ * FROM t) GROUP BY k",
        "Q2 sum(CAST(d8 AS DECIMAL(30, 2)))" ->
          "SELECT sum(CAST(d8 AS DECIMAL(30, 2))) FROM t",
        "Q3 avg(d8)" -> "SELECT avg(d8) FROM t",
        s"Q4 sum(d8) $window" -> s"SELECT rid, sum(d8) $window AS s FROM t",
        s"Q4b avg(d8) $window" -> s"SELECT rid, avg(d8) $window AS a FROM t",
        s"Q4c sum(CAST(d8 AS DECIMAL(30, 2))) $window" ->
          s"SELECT rid, sum(CAST(d8 AS DECIMAL(30, 2))) $window AS s FROM t",
        "X1 sum(d18) (no DecimalAggregates rewrite)" -> "SELECT sum(d18) FROM t",
        "X2 avg(d18) (no DecimalAggregates rewrite)" -> "SELECT avg(d18) FROM t",
        "X3 sum(d8 * d9) (arithmetic in an aggregate's input)" ->
          "SELECT sum(d8 * d9) FROM t",
        "X4 d8 + d9 as a grouping key" ->
          "SELECT d8 + d9 AS g, count(*) FROM t GROUP BY d8 + d9",
        "X5 sum(d8 + d9) over a filter" ->
          "SELECT sum(d8 + d9) FROM t WHERE d8 * d9 > 0")
      queries.foreach { case (title, q) => report(title, spark.sql(q), _.sql(q)) }

      val session = spark
      import session.implicits._
      report("D1 Dataset[java.math.BigDecimal] map(x => x.multiply(x))",
        spark.table("t").select($"d8").as[java.math.BigDecimal]
          .map(x => if (x == null) null else x.multiply(x))(Encoders.DECIMAL),
        s => {
          import s.implicits._
          s.table("t").select($"d8").as[java.math.BigDecimal]
            .map(x => if (x == null) null else x.multiply(x))(Encoders.DECIMAL)
        })
      report("D2 Dataset[scala BigDecimal] map(x => x + x)",
        spark.table("t").select($"d8").as[BigDecimal]
          .map(x => if (x == null) null else x + x),
        s => {
          import s.implicits._
          s.table("t").select($"d8").as[BigDecimal].map(x => if (x == null) null else x + x)
        })
      val rowSchema = spark.table("t").select("d8").schema
      report("D5 Dataset[Row] map(identity)(Encoders.row(d8 DECIMAL(8, 2)))",
        spark.table("t").select("d8").map(r => r)(Encoders.row(rowSchema)),
        s => s.table("t").select("d8").map(r => r)(Encoders.row(rowSchema)))
      val local = Seq(BigDecimal("1.25"), BigDecimal("2.50")).toDS()
      report("D3 Seq[BigDecimal].toDS() (encoded on the driver)", local.toDF(),
        s => { import s.implicits._; Seq(BigDecimal("1.25"), BigDecimal("2.50")).toDS().toDF() })
      report("D4 Dataset[BigDecimal] select(value + value) (SQL over the encoded column)",
        local.selectExpr("value + value"),
        s => {
          import s.implicits._
          Seq(BigDecimal("1.25"), BigDecimal("2.50")).toDS().selectExpr("value + value")
        })
    } finally {
      spark.stop()
    }
  }
}
