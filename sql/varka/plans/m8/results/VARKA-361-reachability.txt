filter (Q4) a FilterExec's condition above the fused inferred null tests, the two places a
fused product would run. Q6 shows coercion's cast of the BIGINT to DECIMAL(20, 0) under the

package org.apache.spark.sql.execution.benchmark

import org.apache.spark.SparkFunSuite
import org.apache.spark.sql.SparkSession
import org.apache.spark.sql.execution.{FilterExec, ProjectExec, SparkPlan, VarkaFilterExec,
  VarkaProjectExec}
import org.apache.spark.sql.execution.aggregate.BaseAggregateExec

class Varka361ReachabilitySuite extends SparkFunSuite {
  test("where q6's product runs") {
    val spark: SparkSession =
      VarkaArrowSessions.createSession("varka-361-reachability", true, false, "local[1]")
    try {
      spark.sql(VarkaDecimalBaselineBenchmark.tableSql(1000)).createOrReplaceTempView("t")
      VarkaArrowSessions.cache(spark, "t")
      val queries = Seq(
        "Q1 sum(x15 * y15)" -> "SELECT sum(x15 * y15) AS s FROM t",
        "Q2 sum(x15 * y15) WHERE z15 > 0" -> "SELECT sum(x15 * y15) AS s FROM t WHERE z15 > 0",
        "Q3 x15 * y15 as a value" -> "SELECT x15 * y15 AS a FROM t",
        "Q4 x15 * y15 < z15 as a filter" -> "SELECT x15 FROM t WHERE x15 * y15 < z15",
        "Q5 sum(x15 * (1 - y15))" -> "SELECT sum(x15 * (1 - y15)) AS s FROM t",
        "Q6 l * x18 as a value" -> "SELECT l * x18 AS a FROM t")
      for ((name, q) <- queries) {
        val plan = spark.sql(q).queryExecution.executedPlan
        // scalastyle:off println
        println(s"==== $name: $q")
        println(plan.treeString)
        def holders(p: SparkPlan): Seq[String] = p.collect {
          case a: BaseAggregateExec if a.aggregateExpressions.exists(_.toString.contains("*")) =>
            "aggregate " + a.nodeName + ": " + a.aggregateExpressions.mkString(", ")
          case f: FilterExec if f.condition.toString.contains("*") =>
            "FilterExec: " + f.condition
          case pr: ProjectExec if pr.projectList.exists(_.toString.contains("*")) =>
            "ProjectExec: " + pr.projectList.mkString(", ")
          case v: VarkaProjectExec => "VarkaProjectExec"
          case v: VarkaFilterExec => "VarkaFilterExec"
        }
        println("evaluated by: " + holders(plan).mkString(" | "))
        // scalastyle:on println
      }
    } finally {
      spark.stop()
    }
  }
}
